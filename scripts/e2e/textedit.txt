# A typed end-to-end run through the BUILT app. Test only: the TEST agent, no microphone, no sound.
# Run:  scripts/live_lock.sh acquire <you> && LIVE_LOCK_OWNER=<you> scripts/run_app.sh --wait --script scripts/e2e/textedit.txt; scripts/live_lock.sh release <you>
# (run_app.sh refuses a script unless LIVE_LOCK_OWNER holds the live lock, and keeps the lock fresh while it runs.)
# Report: ~/Library/Application Support/Handsfree/script-run.json
# It moves the real pointer and types into a new, unsaved TextEdit document, then closes it without saving.

require-unlocked

wake open TextEdit
expect-tool open_app
expect-reply
expect-front TextEdit

say Make a new document and type: hello from Handsfree
expect-tool start_task
wait-task 180
expect-text TextEdit | hello from Handsfree

say What did you just do?
expect-reply
expect-no-tool
expect-reply-without tool,function

say That's all.
wait-asleep 20

# Only throws the document away if TextEdit was not running before and it holds exactly this text.
close-without-saving TextEdit | hello from Handsfree
